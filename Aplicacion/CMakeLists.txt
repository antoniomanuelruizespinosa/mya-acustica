cmake_minimum_required(VERSION 3.22)
project(MyaAcustica VERSION 0.3.3 LANGUAGES C CXX)
add_compile_definitions(MYA_VERSION_STRING="${PROJECT_VERSION}")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED JUCE_DIR)
 message(FATAL_ERROR "Set JUCE_DIR to the JUCE 8.0.12 source directory")
endif()
add_subdirectory(${JUCE_DIR} juce)
# Keep the technical product/executable and preference locations; brand display separately.
set(MYA_DISPLAY_NAME "Mya Acústica Analyzer")
juce_add_gui_app(MyaAcustica PRODUCT_NAME "Mya Acustica" PLIST_TO_MERGE "<plist><dict><key>CFBundleName</key><string>${MYA_DISPLAY_NAME}</string><key>CFBundleDisplayName</key><string>${MYA_DISPLAY_NAME}</string></dict></plist>" COMPANY_NAME "Mya Acustica" ICON_BIG "assets/branding/mya-analyzer-icon-1024.png" ICON_SMALL "assets/branding/mya-analyzer-icon-128.png" BUNDLE_ID "com.mya.acustica.prototype" MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "Mya Acústica Analyzer usa las entradas seleccionadas solo al pulsar Play para mostrar niveles digitales y SPL RMS calibrado.")
juce_add_binary_data(MyaBrandAssets HEADER_NAME MyaBrandAssets.h NAMESPACE MyaBrandAssets SOURCES assets/branding/mya-analyzer-icon-v1.png)
target_sources(MyaAcustica PRIVATE src/Main.cpp)
if(APPLE)
 target_sources(MyaAcustica PRIVATE src/NativeTitleBranding_mac.mm)
endif()
target_compile_definitions(MyaAcustica PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_APPLICATION_NAME_STRING="${MYA_DISPLAY_NAME}" JUCE_APPLICATION_VERSION_STRING="${PROJECT_VERSION}")
target_link_libraries(MyaAcustica PRIVATE MyaBrandAssets juce::juce_audio_devices juce::juce_gui_extra juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
add_executable(mya_migrate_data tools/migrate-data.cpp)
target_link_libraries(mya_migrate_data PRIVATE juce::juce_core)

# ASIO requires Steinberg's SDK when producing the Windows build.
option(MYA_ENABLE_ASIO "Enable ASIO backend on Windows" OFF)
set(MYA_ASIO_SDK_DIR "" CACHE PATH "Steinberg ASIO SDK directory")
if(WIN32 AND MYA_ENABLE_ASIO)
 if(NOT EXISTS "${MYA_ASIO_SDK_DIR}/common/iasiodrv.h")
  message(FATAL_ERROR "MYA_ENABLE_ASIO requires MYA_ASIO_SDK_DIR with common/iasiodrv.h")
 endif()
 target_compile_definitions(MyaAcustica PRIVATE JUCE_ASIO=1)
 target_include_directories(MyaAcustica PRIVATE "${MYA_ASIO_SDK_DIR}/common")
endif()
