cmake_minimum_required(VERSION 3.22)
project(MyaAcustica VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED JUCE_DIR)
 message(FATAL_ERROR "Set JUCE_DIR to the JUCE 8.0.12 source directory")
endif()
add_subdirectory(${JUCE_DIR} juce)
juce_add_gui_app(MyaAcustica PRODUCT_NAME "Mya Acustica" COMPANY_NAME "Mya Acustica" BUNDLE_ID "com.mya.acustica.prototype")
target_sources(MyaAcustica PRIVATE src/Main.cpp)
target_compile_definitions(MyaAcustica PRIVATE MYA_PROJECT_DIRECTORY="${CMAKE_CURRENT_SOURCE_DIR}" JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_APPLICATION_NAME_STRING="$<TARGET_PROPERTY:MyaAcustica,JUCE_PRODUCT_NAME>" JUCE_APPLICATION_VERSION_STRING="0.3.0")
target_link_libraries(MyaAcustica PRIVATE juce::juce_audio_devices juce::juce_gui_extra juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)

# ASIO requires Steinberg's SDK when producing the Windows build.
option(MYA_ENABLE_ASIO "Enable ASIO backend on Windows" OFF)
set(MYA_ASIO_SDK_DIR "" CACHE PATH "Steinberg ASIO SDK directory")
if(WIN32 AND MYA_ENABLE_ASIO)
 if(NOT EXISTS "${MYA_ASIO_SDK_DIR}/common/iasiodrv.h")
  message(FATAL_ERROR "MYA_ENABLE_ASIO requires MYA_ASIO_SDK_DIR with common/iasiodrv.h")
 endif()
 target_compile_definitions(MyaAcustica PRIVATE JUCE_ASIO=1)
 target_include_directories(MyaAcustica PRIVATE "${MYA_ASIO_SDK_DIR}/common")
endif()

include(CTest)
option(MYA_BUILD_LAYOUT_SNAPSHOT "Build offline component image renderer" OFF)
if(MYA_BUILD_LAYOUT_SNAPSHOT)
 juce_add_console_app(mya_layout_snapshot PRODUCT_NAME "Mya layout snapshot")
 target_sources(mya_layout_snapshot PRIVATE tests/layout_snapshot.cpp)
 target_compile_definitions(mya_layout_snapshot PRIVATE MYA_PROJECT_DIRECTORY="${CMAKE_CURRENT_BINARY_DIR}/snapshot-fixture" JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 target_link_libraries(mya_layout_snapshot PRIVATE juce::juce_audio_devices juce::juce_gui_extra juce::juce_recommended_config_flags)
endif()
if(BUILD_TESTING)
 add_executable(mya_smoothing_test tests/smoothing.cpp)
 add_executable(mya_presentation_test tests/presentation.cpp)
 add_executable(mya_commands_test tests/commands.cpp)
 add_executable(mya_library_test tests/library.cpp)
 add_executable(mya_impulse_test tests/impulse.cpp)
 add_executable(mya_ui_layout_test tests/ui_layout.cpp)
 add_executable(mya_preferences_test tests/preferences.cpp)
 add_executable(mya_input_meters_test tests/input_meters.cpp)
 target_link_libraries(mya_input_meters_test PRIVATE juce::juce_core)
 target_compile_definitions(mya_input_meters_test PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 target_link_libraries(mya_preferences_test PRIVATE juce::juce_core)
 target_compile_definitions(mya_preferences_test PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 target_link_libraries(mya_ui_layout_test PRIVATE juce::juce_core)
 target_compile_definitions(mya_ui_layout_test PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 target_link_libraries(mya_library_test PRIVATE juce::juce_core)
 target_compile_definitions(mya_library_test PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 # Assertions are the test oracle, including in Release configurations.
 if(MSVC)
  target_compile_options(mya_smoothing_test PRIVATE /UNDEBUG)
  target_compile_options(mya_presentation_test PRIVATE /UNDEBUG)
  target_compile_options(mya_commands_test PRIVATE /UNDEBUG)
  target_compile_options(mya_library_test PRIVATE /UNDEBUG)
  target_compile_options(mya_impulse_test PRIVATE /UNDEBUG)
  target_compile_options(mya_ui_layout_test PRIVATE /UNDEBUG)
  target_compile_options(mya_preferences_test PRIVATE /UNDEBUG)
  target_compile_options(mya_input_meters_test PRIVATE /UNDEBUG)
 else()
  target_compile_options(mya_smoothing_test PRIVATE -UNDEBUG)
  target_compile_options(mya_presentation_test PRIVATE -UNDEBUG)
  target_compile_options(mya_commands_test PRIVATE -UNDEBUG)
  target_compile_options(mya_library_test PRIVATE -UNDEBUG)
  target_compile_options(mya_impulse_test PRIVATE -UNDEBUG)
  target_compile_options(mya_ui_layout_test PRIVATE -UNDEBUG)
  target_compile_options(mya_preferences_test PRIVATE -UNDEBUG)
  target_compile_options(mya_input_meters_test PRIVATE -UNDEBUG)
 endif()
 add_test(NAME input_meters COMMAND mya_input_meters_test "${CMAKE_CURRENT_BINARY_DIR}/input-meters-test-data")
 add_test(NAME smoothing COMMAND mya_smoothing_test)
 add_test(NAME presentation COMMAND mya_presentation_test)
 add_test(NAME commands COMMAND mya_commands_test)
 add_test(NAME library COMMAND mya_library_test "${CMAKE_CURRENT_BINARY_DIR}/library-test-data")
 add_test(NAME impulse COMMAND mya_impulse_test)
 add_test(NAME ui_layout COMMAND mya_ui_layout_test)
 add_test(NAME preferences COMMAND mya_preferences_test "${CMAKE_CURRENT_BINARY_DIR}/preferences-test-data")
endif()
